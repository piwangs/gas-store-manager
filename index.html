-- 1. Enable Required Extensions CREATE EXTENSION IF NOT EXISTS pgcrypto; -- 2. Create Locations Table CREATE TABLE IF NOT EXISTS locations ( id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY, name TEXT UNIQUE NOT NULL ); -- 3. Create Users Table CREATE TABLE IF NOT EXISTS store_users ( id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY, username TEXT UNIQUE NOT NULL, password TEXT NOT NULL, role TEXT NOT NULL CHECK (role IN ('Manager', 'Storekeeper')), location TEXT NOT NULL, name TEXT NOT NULL, disabled BOOLEAN DEFAULT false ); -- 4. Create Daily Inventory Table with all required columns CREATE TABLE IF NOT EXISTS inventory_entries ( id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY, location TEXT NOT NULL, entry_date DATE NOT NULL, item_code TEXT NOT NULL, description TEXT NOT NULL, category TEXT NOT NULL, unit TEXT NOT NULL, opening NUMERIC DEFAULT 0, added NUMERIC DEFAULT 0, issues NUMERIC DEFAULT 0, cost NUMERIC DEFAULT 0, price NUMERIC DEFAULT 0, comments TEXT, updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(), UNIQUE(location, entry_date, item_code) ); -- 5. Create Audit Logs Table CREATE TABLE IF NOT EXISTS audit_logs ( id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY, created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(), username TEXT, role TEXT, location TEXT, action TEXT, details TEXT ); -- 6. Secure Credentials Verification Function CREATE OR REPLACE FUNCTION verify_store_user(input_username TEXT, input_password TEXT) RETURNS SETOF store_users LANGUAGE plpgsql SECURITY DEFINER AS $$ BEGIN RETURN QUERY SELECT * FROM store_users WHERE username = input_username AND password = input_password AND disabled = false; END; $$; -- 7. Secure Upsert Function enforcing Storekeeper vs Manager privileges CREATE OR REPLACE FUNCTION secure_upsert_inventory( p_username TEXT, p_location TEXT, p_entry_date DATE, p_item_code TEXT, p_description TEXT, p_category TEXT, p_unit TEXT, p_opening NUMERIC, p_added NUMERIC, p_issues NUMERIC, p_cost NUMERIC, p_price NUMERIC, p_comments TEXT ) RETURNS VOID LANGUAGE plpgsql SECURITY DEFINER AS $$ DECLARE v_user_role TEXT; v_user_location TEXT; v_user_disabled BOOLEAN; v_existing_id BIGINT; BEGIN SELECT role, location, disabled INTO v_user_role, v_user_location, v_user_disabled FROM store_users WHERE username = p_username; IF NOT FOUND OR v_user_disabled THEN RAISE EXCEPTION 'Access Denied: Invalid or disabled user account.'; END IF; IF v_user_role = 'Storekeeper' AND v_user_location <> p_location THEN RAISE EXCEPTION 'Access Denied: Storekeepers can only edit their assigned location.'; END IF; -- Check if entry already exists SELECT id INTO v_existing_id FROM inventory_entries WHERE location = p_location AND entry_date = p_entry_date AND item_code = p_item_code; -- If Storekeeper tries to create a brand new item definition, block it (Storekeepers only log added/issues/comments on existing items) IF v_user_role = 'Storekeeper' AND v_existing_id IS NULL THEN RAISE EXCEPTION 'Permission Denied: Storekeepers cannot create new item catalog definitions. Contact a Manager.'; END IF; IF v_user_role = 'Storekeeper' THEN -- Storekeepers can ONLY update added, issues, and comments on existing rows UPDATE inventory_entries SET added = p_added, issues = p_issues, comments = p_comments, updated_at = NOW() WHERE location = p_location AND entry_date = p_entry_date AND item_code = p_item_code; else -- Managers have full rights to insert/update all fields INSERT INTO inventory_entries ( location, entry_date, item_code, description, category, unit, opening, added, issues, cost, price, comments, updated_at ) VALUES ( p_location, p_entry_date, p_item_code, p_description, p_category, p_unit, p_opening, p_added, p_issues, p_cost, p_price, p_comments, NOW() ) ON CONFLICT (location, entry_date, item_code) DO UPDATE SET description = EXCLUDED.description, category = EXCLUDED.category, unit = EXCLUDED.unit, opening = EXCLUDED.opening, added = EXCLUDED.added, issues = EXCLUDED.issues, cost = EXCLUDED.cost, price = EXCLUDED.price, comments = EXCLUDED.comments, updated_at = NOW(); END IF; END; $$; -- 8. Low Stock Report Function CREATE OR REPLACE FUNCTION get_low_stock_report(p_location TEXT, p_threshold NUMERIC DEFAULT 5) RETURNS TABLE ( item_code TEXT, description TEXT, category TEXT, closing_stock NUMERIC, unit TEXT ) LANGUAGE plpgsql SECURITY DEFINER AS $$ BEGIN RETURN QUERY SELECT i.item_code, i.description, i.category, (i.opening + i.added - i.issues) AS closing_stock, i.unit FROM inventory_entries i WHERE i.location = p_location AND i.entry_date = CURRENT_DATE AND (i.opening + i.added - i.issues) <= p_threshold; END; $$; -- 9.
